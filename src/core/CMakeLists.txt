cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(pure_nav_core LANGUAGES CXX)
    option(PURE_NAV_BUILD_TESTS "构建核心接口合同测试" ON)
    if(PURE_NAV_BUILD_TESTS)
        enable_testing()
    endif()
endif()

find_package(Eigen3 REQUIRED)
add_library(pure_nav_core INTERFACE)
target_compile_features(pure_nav_core INTERFACE cxx_std_17)
target_include_directories(pure_nav_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(pure_nav_core INTERFACE Eigen3::Eigen)

if(PURE_NAV_BUILD_TESTS)
    add_executable(test_core_contracts ../test/test_core/test_core.cpp)
    target_link_libraries(test_core_contracts PRIVATE pure_nav_core)
    add_test(NAME core_contracts COMMAND test_core_contracts)
endif()
